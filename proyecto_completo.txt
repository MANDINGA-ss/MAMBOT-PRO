async def on_message_handler(bot, message):
    if message.author.bot:
        return
 
    canal_id = str(message.channel.id)
    user_id = str(message.author.id)
    contenido_lower = message.content.lower().strip()
 
    # ================= 1. COMANDOS DIRECTOS =================
    # Primero procesamos comandos para que !reset, !perfil, etc. funcionen siempre
    await bot.process_commands(message)
 
    if contenido_lower.startswith("!lol"):
        await manejar_lol(message)
        return
 
    # Ignorar cualquier otro comando para no procesarlo como mensaje normal
    if contenido_lower.startswith("!"):
        return
 
    # ================= 2. ACTUALIZACIÃ“N DE PERFIL =================
    # actualizar_perfil ya incrementa mensajes internamente, no duplicar
    actualizar_perfil(message.author)
 
    # Cada 15 mensajes analizamos personalidad (usando el contador ya actualizado)
    if perfiles.get(user_id, {}).get("mensajes", 0) % 15 == 0:
        p = await analizar_personalidad(user_id)
        if p:
            aplicar_estilo(user_id, p)
 
    # ================= 3. MEMORIA =================
    memoria_canales[canal_id].append({
        "role": "user",
        "content": message.content,
        "name": message.author.name
    })
 
    await reaccionar_si_corresponde(message)
 
    # ================= 4. RESPUESTA DE LA IA =================
    contenido = message.content.lower()
    disparadores = ["bot", "ia", "ayuda", "mambot"]  # Quitado "?" â€” demasiado amplio
 
    if bot.user not in message.mentions and not any(d in contenido for d in disparadores):
        guardar_canales()
        guardar_usuarios()
        return
 
    mensajes = [
        {
            "role": "system",
            "content": (
                "Tu nombre es MAMBOT (tambiÃ©n llamado mambot). "
                "Sos una IA argentina que vive en un servidor de Discord informal, sos como un usuario mas. "

                "AdaptÃ¡s el nivel de humor segÃºn el contexto y el usuario. "

def crear_embed_lol(datos):
    """

        "nombre": str,
        "tag": str,
        "tier": str,

async def manejar_lol(message):
    contenido = message.content.strip()
    contenido_lower = contenido.lower()

    # Delegar vincular a su propio handler â€” formato: !lol vincular Nombre#TAG
    if "vincular" in contenido_lower:
        await vincular_lol(message)
        return

    partes = contenido.split(" ", 1)

    if len(partes) < 2 or "#" not in partes[1]:
        await message.channel.send("âŒ UsÃ¡: `!lol Nombre#TAG` (Ej: `!lol Faker#KR1`)")
        return

    game_name, tag = partes[1].split("#", 1)
    msg_cargando = await message.channel.send(f"ðŸ” Buscando a **{game_name}#{tag}**...")

    try:
        resultado = await obtener_elo(game_name, tag)

        if resultado is None:
            await msg_cargando.edit(content="âŒ No encontrÃ© la cuenta. RevisÃ¡ el Nombre#TAG y que sea de LAS (la2).")
            return

        league_data = resultado.get("league", [])

        # Buscar SoloQ primero, si no Flex
        stats_rank = next((l for l in league_data if l.get("queueType") == "RANKED_SOLO_5x5"), None)
        if not stats_rank:
            stats_rank = next((l for l in league_data if l.get("queueType") == "RANKED_FLEX_SR"), None)

        if not stats_rank:
            await msg_cargando.edit(content=f"ðŸŽ® **{game_name}#{tag}** no tiene partidas rankeadas (SoloQ/Flex) en esta regiÃ³n.")
            return

        datos_finales = {
            "nombre": game_name,
            "tag": tag,
            "tier": stats_rank.get("tier", "UNRANKED"),
            "rank": stats_rank.get("rank", ""),
            "lp": stats_rank.get("leaguePoints", 0),
            "wins": stats_rank.get("wins", 0),
            "losses": stats_rank.get("losses", 0),
            "queue": "SoloQ" if stats_rank.get("queueType") == "RANKED_SOLO_5x5" else "Flex",
            "profile_icon_url": resultado.get("profile_icon_url"),
            "summoner_level": resultado.get("summoner_level"),
            "campeon_nombre": resultado.get("campeon_nombre"),
            "campeon_imagen": resultado.get("campeon_imagen"),
            "campeon_puntos": resultado.get("campeon_puntos"),
        }

        embed = crear_embed_lol(datos_finales)
        await msg_cargando.delete()
        await message.channel.send(embed=embed)

    except Exception as e:
        print(f"âŒ Error en handler de LoL: {e}")
        try:
            await msg_cargando.edit(content="âŒ Hubo un fallo interno al procesar los datos de Riot.")

import aiohttp
import os
import urllib.parse

async def obtener_version_ddragon(session):
    url = "https://ddragon.leagueoflegends.com/api/versions.json"
    async with session.get(url) as r:
        if r.status == 200:
            versions = await r.json()
            return versions[0]
    return "15.1.1"  # fallback

async def obtener_nombre_campeon(session, champion_id, version):
    url = f"https://ddragon.leagueoflegends.com/cdn/{version}/data/es_AR/champion.json"
    async with session.get(url) as r:
        if r.status != 200:
            return None, None
        data = await r.json()
        for nombre, info in data["data"].items():
            if int(info["key"]) == champion_id:
                imagen = f"https://ddragon.leagueoflegends.com/cdn/{version}/img/champion/{info['image']['full']}"
                return info["name"], imagen
    return None, None

async def obtener_elo(game_name: str, tag: str):
    api_key = os.getenv("RIOT_API_KEY")
    headers = {"X-Riot-Token": api_key}

    name_encoded = urllib.parse.quote(game_name.strip())
    tag_encoded = urllib.parse.quote(tag.strip())

    async with aiohttp.ClientSession() as session:
        try:
            # 1. Account-V1 (americas) â€” PUUID
            url_acc = f"https://americas.api.riotgames.com/riot/account/v1/accounts/by-riot-id/{name_encoded}/{tag_encoded}"
            async with session.get(url_acc, headers=headers) as r:
                if r.status != 200:
                    print(f"âš ï¸ Account-V1 fallÃ³: {r.status}")
                    return None
                account = await r.json()
                puuid = account.get("puuid")

            # 2. Summoner-V4 â€” Ã­cono y nivel
            url_sum = f"https://la2.api.riotgames.com/lol/summoner/v4/summoners/by-puuid/{puuid}"
            async with session.get(url_sum, headers=headers) as r:
                summoner = await r.json() if r.status == 200 else {}
                profile_icon_id = summoner.get("profileIconId")
                summoner_level = summoner.get("summonerLevel")

            # 3. League-V4 â€” elo
            url_league = f"https://la2.api.riotgames.com/lol/league/v4/entries/by-puuid/{puuid}"
            async with session.get(url_league, headers=headers) as r:
                if r.status != 200:
                    print(f"âš ï¸ League-V4 fallÃ³: {r.status}")
                    return None
                league_data = await r.json()

            # 4. Champion-Mastery-V4 â€” campeÃ³n mÃ¡s jugado
            url_mastery = f"https://la2.api.riotgames.com/lol/champion-mastery/v4/champion-masteries/by-puuid/{puuid}/top?count=1"
            async with session.get(url_mastery, headers=headers) as r:
                mastery_data = await r.json() if r.status == 200 else []

            # 5. Data Dragon â€” versiÃ³n actual y nombre/imagen del campeÃ³n
            version = await obtener_version_ddragon(session)
            profile_icon_url = f"https://ddragon.leagueoflegends.com/cdn/{version}/img/profileicon/{profile_icon_id}.png" if profile_icon_id else None

            campeon_nombre = None
            campeon_imagen = None
            campeon_puntos = None
            if mastery_data:
                champion_id = mastery_data[0].get("championId")
                campeon_puntos = mastery_data[0].get("championPoints", 0)
                campeon_nombre, campeon_imagen = await obtener_nombre_campeon(session, champion_id, version)

            return {
                "league": league_data,
                "profile_icon_url": profile_icon_url,
                "summoner_level": summoner_level,
                "campeon_nombre": campeon_nombre,
                "campeon_imagen": campeon_imagen,
                "campeon_puntos": campeon_puntos,
            }

        except Exception as e:
            print(f"ðŸ”¥ Error en stats: {e}")
            return None

import aiohttp
import random
import re
import discord
from config import GIPHY_API_KEY
 
async def buscar_gif(query):
    try:
        url = f"https://api.giphy.com/v1/gifs/search?api_key={GIPHY_API_KEY}&q={query}&limit=10"
        async with aiohttp.ClientSession() as session:
            async with session.get(url, timeout=aiohttp.ClientTimeout(total=5)) as resp:
                if resp.status != 200:
                    return None
                data = await resp.json()
                if data.get("data"):
                    return random.choice(data["data"])["images"]["original"]["url"]
    except Exception as e:
        print(f"âš ï¸ Error buscando GIF: {e}")
    return None
 
async def enviar_con_gif(canal, texto):
    match = re.search(r"\[GIF:\s*(.*?)\]", texto)
    if not match:
        await canal.send(texto)
        return
 
    query = match.group(1)
    texto_limpio = re.sub(r"\[GIF:.*?\]", "", texto).strip()
 
    gif = await buscar_gif(query)
 
    embed = discord.Embed(description=texto_limpio)
    if gif:
        embed.set_image(url=gif)
 
    await canal.send(embed=embed)

import json
from pathlib import Path

DATA_PATH = Path("data/profiles.json")
perfiles = {}

def cargar_perfiles():
    """Carga los perfiles desde el JSON y normaliza las claves faltantes."""
    global perfiles
    if DATA_PATH.exists() and DATA_PATH.stat().st_size > 0:
        with open(DATA_PATH, "r", encoding="utf-8") as f:
            perfiles.update(json.load(f))
    else:
        perfiles.clear()

    # Normalizar perfiles existentes para que no falten claves
    for user_id, perfil in perfiles.items():
        perfil.setdefault("nombre", f"Usuario {user_id}")
        perfil.setdefault("mensajes", 0)
        perfil.setdefault("nivel", 1)
        perfil.setdefault("rol", "Nuevo")
        perfil.setdefault("estilo", "Indefinido")
        perfil.setdefault("lol", None)

def guardar():
    """Guarda los perfiles en el JSON."""
    DATA_PATH.parent.mkdir(exist_ok=True)
    with open(DATA_PATH, "w", encoding="utf-8") as f:
        json.dump(perfiles, f, indent=4, ensure_ascii=False)

def actualizar_perfil(user):
    """Actualiza el perfil del usuario, sube nivel y asigna rol segÃºn actividad."""
    user_id = str(user.id)

    # Si el perfil no existe, crear con valores por defecto
    if user_id not in perfiles:
        perfiles[user_id] = {
            "nombre": user.name,
            "mensajes": 0,
            "nivel": 1,
            "rol": "Nuevo",
            "estilo": "Indefinido",
            "lol": None
        }

    # Normalizar claves por si falta algo en perfiles antiguos
    perfil = perfiles[user_id]
    perfil.setdefault("mensajes", 0)
    perfil.setdefault("nivel", 1)
    perfil.setdefault("rol", "Nuevo")
    perfil.setdefault("estilo", "Indefinido")
    perfil.setdefault("lol", None)

    # Actualizar mensajes
    perfil["mensajes"] += 1

    # Subir nivel cada 30 mensajes
    if perfil["mensajes"] % 30 == 0:
        perfil["nivel"] += 1

    # Asignar rol segÃºn nivel
    lvl = perfil["nivel"]
    if lvl >= 10:
        perfil["rol"] = "Veterano"
    elif lvl >= 5:
        perfil["rol"] = "Activo"
    else:
        perfil["rol"] = "Nuevo"

    guardar()

def aplicar_estilo(user_id, personalidad):
    """Aplica un estilo de personalidad al perfil de un usuario."""
    if user_id in perfiles and personalidad:
        perfiles[user_id]["estilo"] = (
            f"{personalidad.get('tono', 'Indefinido')} / "
            f"{personalidad.get('nivel_tecnico', 'Indefinido')} / "
            f"{personalidad.get('humor', 'Indefinido')}"
        )
        guardar()

        return "humor"
    if any(p in t for p in PALABRAS_BARDEO):
        return "bardeo"
    if any(p in t for p in PALABRAS_ENOJO):
        return "enojo"

    return "random"


async def reaccionar_si_corresponde(message) -> bool:
    """
    Devuelve True si reaccionÃ³ (y quizÃ¡ no haga falta responder)
    """

    # probabilidad base
    if random.random() > 0.35:  # 35% de chance
        return False

    tipo = detectar_tipo(message.content)
    emoji = random.choice(REACCIONES.get(tipo, REACCIONES["random"]))

    try:
        await message.add_reaction(emoji)
        return True
    except:
        return False

def cargar_json(path, default):
    if path.exists():
        with open(path, "r", encoding="utf-8") as f:

def guardar_json(path, data):
    with open(path, "w", encoding="utf-8") as f:

async def buscar_gif(query):
    url = f"https://api.giphy.com/v1/gifs/search?api_key={GIPHY_API_KEY}&q={query}&limit=10"
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as resp:
            data = await resp.json()
            if data["data"]:
                return random.choice(data["data"])["images"]["original"]["url"]
    return None

async def enviar_con_gif(canal, texto):
    match = re.search(r"\[GIF:\s*(.*?)\]", texto)
    if match:
        query = match.group(1)
        texto = re.sub(r"\[GIF:.*?\]", "", texto).strip()
        gif = await buscar_gif(query)
        embed = discord.Embed(description=texto)
        if gif:
            embed.set_image(url=gif)
        await canal.send(embed=embed)
    else:
        await canal.send(texto)
